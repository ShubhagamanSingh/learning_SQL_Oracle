join operation :- 

		if you want to select data from two tables then you can  perform join operation.
		for join operation  atleast onr field must be common in both tables.

	imagine tables:-
	
		table_1---> col_11(P.K.), col_12;
		table_2---> col_21(P.K.), col_22, col_11(F.K.);
		
		P.K. => primary key.
		F.K. => foreign key.

	SYNTAX:- select col_12 from table_1 and col_22 from table_2 baised on col_11:-
		
		select col_12, col_22 from table_1, table_2 where table_2.col_11=<value>;

left join operation:-

		when you perform left join operation, then all records of left table are desplayed and matching 
		record of right table are desplayed.
		
	syntax:-
	
		select col_12, col_22 from table_1 left join table_2 on table_1.col_11=table_2.col_11;
		
		E.g.:- select empname, pname from employee left join product on employee.empid=product.empid;

right join operation:-

		when you perform right join operation, then all records of right table are desplayed and matching 
		record of left table are desplayed.
		
	syntax:-
	
		select col_12, col_22 from table_1 right join table_2 on table_1.col_11=table_2.col_11;
		
		E.g.:- select empname, pname from employee right join product on employee.empid=product.empid;

view:-
	view is a logical table which is created by another table.
	if you have a table with some important fields and some unimportant fields,
	so you can create a view of unimportant fields and assign that view to an operator.
	
	first we create a tabel tb1 with (id, name, salary)these fields:-
	
	create table tb1
	(
	id number(6) primary key,
	name varchar2(30),
	salary number(8)
	);
	
	now we create view named tb:-
	
	create view tb as select id,name from tb1;
	

check constrant:-
	check constrant is used for validations in database object;
	
empid varchar2(10) p.k. must start with 'SPI'.
empname varchar2(30) empname must be in upprs case.
country varchar2(5) country must be 'India' or 'Nepal'.
salary number(6) salary not more than 100000.
----
SQL> create table spiemp
  2  (
  3  empid varchar2(10) check(empid like 'SPI%'),
  4  empname varchar2(30) check(empname=upper(empname)),
  5  country varchar2(5) check(country in ('India','Nepal')),
  6  salary number(6) check(salary<=100000),
  7  primary key(empid)
  8  );

Table created.

SQL> desc spiemp;

 ----------------------------------------- -------- ----------------------------

		1. BASIC LOOP;

		3. FOR LOOP;
		
BASIC LOOP:-
		
	SYNTAX:-

PL/SQL CODE TO PRINT NUMBER 1 TO 5:-

declare
a int;
begin
a:=1;
loop
dbms_output.put_line(a);
a:=a+1;
exit when a>5;
end loop;
end;
/

WHILE LOOP:-
		
	SYNTAX:-
		
	while(condition)
	loop
	/* code */
	end loop;					

PL/SQL CODE TO PRINT TABLE OF GIVEN NUMBER:-

declare
n int;
i int;
begin
n:=&n;
i:=1;
while(n<=10)
loop
dbms_output.put_line(n||'*'||i||'='||(n*i));
i:=i+1;
end loop;
end;
/

FOR LOOP:-

	syntax:-
	
	for <loop variable> in <lower limit>..<upper limit>
	loop
	/* code */
	end loop;
	end;
