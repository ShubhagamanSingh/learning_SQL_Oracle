JOIN OPERATION :- 

		IF YOU WANT TO SELECT DATA FROM TWO TABLES THEN CAN YOU PERFORM JOIN OPERATION.
		FOR JOIN OPERATION ATLEAST ONE FIELD MUST BE COMMON IN BOTH TABLES.

	TABLES:-
	
		table_1---> col_11(P.K.), col_12;
		table_2---> col_21(P.K.), col_22, col_11(F.K.);

	SYNTAX:-
	
		select col_12 fom table_1 and col_22 from table_2 baised on col_11:-
		
		select col_12, col_22 from table_1, table_2 where table_1.col_11=table_2.col_11;
		
		E.g.:- select empname, pname from employee, product where employee.empid=product.empid;

____________________________________________________________________________________________|	

//
	employee--->empid(P.K.), empname;
	product--->pid(P.K.), pname, empid(F.K.); //

LEFT JOIN OPERATION :- 

		WHEN YOU PERFOEM LEFT JOIN OPERATION, THEN ALL RECORD OF LEFT TABLE ARE DISPLAYED
		AND MATCHING RECORD OF RIGHT TABLE ARE DISPLAYED.
		
TABLES:-
	
		table_1---> col_11(P.K.), col_12;
		table_2---> col_21(P.K.), col_22, col_11(F.K.);

SYNTAX:-

		select col_12, col_22 from table_1 left join table_2 on table_1.col_11=table_2.col_11;
		
		E.g.:- select empname, pname from employee left join product on employee.empid=product.empid;

RIGHT JOIN OPERATION :- 

		WHEN YOU PERFOEM RIGHT JOIN OPERATION, THEN ALL RECORD OF RIGHT TABLE ARE DISPLAYED

TABLES:-
	
		table_1---> col_11(P.K.), col_12;

SYNTAX:-

		select col_12, col_22 from table_1 right join table_2 on table_1.col_11=table_2.col_11;
		
		E.g.:- select empname, pname from employee right join product on employee.empid=product.empid;

VIEW:-
	
	VIEW IS A LOGICAL TABLE WHICH IS CREATED BY ANOTHER TABLE.
	IF YOU HAVE A TABLE WITH SOME IMPORTANT FIELDS AND SOME UNIMPORTENT FIELDS, 
	SO YOU CAN CREATE A VIEW OF UNIMPORTANT FIELDS AND ASSIGN THAT VIEW TO AN OPERATOR.
	WHEN OPERATOR INSERT RECORDS IN VIEW THEN MAIN TABLE WILL ALSO AFFECTED.
