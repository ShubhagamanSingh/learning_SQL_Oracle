a int;
begin
a:=&a;
if a mod 2=0 then
dbms_output.put_line('Even');
else

end;

LOOP IN PL/SQL:-

LOOP CONTROL:-
		
		IF YOU HAVE A BLOCK OF CODE WHICH YOU WANT TO REPEATEDLY UP TO GIVEN CONDOTION IS TRUE THEN YOU CAN USE LOOP CONTROL.
		
		IN PL/SQL THERE ARE THREE TYPES OF LOOP CONTROL:-
		
		1. BASIC LOOP;
		2. WHILE LOOP;
		3. FOR LOOP;
		
BASIC LOOP:-
		
	SYNTAX:-
			
	loop
	/* code */
	end loop;
	
PL/SQL CODE TO PRINT NUMBER 1 TO 5:-

declare
a int;
begin
a:=1;
loop
dbms_output.put_line(a);
a:=a+1;
exit when a>5;
end loop;
end;
/

WHILE LOOP:-
		
		WHILE LOOP IS WORK LIKE ENTRY CONTROL. IN WHILE LOOP FIRST CONDITION IS CHECKED THEN CODE WILL EXECUTED.
		
	SYNTAX:-
		
	while(condition)
	loop
	/* code */
	end loop;					

PL/SQL CODE TO PRINT TABLE OF GIVEN NUMBER:-

declare
n int;
i int;
begin
n:=&n;
i:=1;
while(n<=10)
loop
dbms_output.put_line(n||'*'||i||'='||(n*i));
i:=i+1;
end loop;
end;
/

FOR LOOP:-
		
		FOR LOOP STATEMENT IS BEST SUTABLE WHEN YOU WANT TO EXECUTE A CODE FOR A KNOWN NUMBER OF TIME RATHER THAN BAISED ON SOME OTHER CONDITION.
		
	SYNTAX:-
	
	for <loop variable> in <lower limit>..<upper limit>
	loop
	/* code */
	end loop;
	end;
	/
